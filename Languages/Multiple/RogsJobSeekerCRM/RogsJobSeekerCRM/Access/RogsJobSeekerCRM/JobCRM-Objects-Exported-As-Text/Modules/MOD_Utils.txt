Sub CreateTypes()
'Created 21/08/2024 By Roger Williams
'
'creates a text file in: C:\Temp
'that contains types structures for each table
'
'
Dim tblTemp As DAO.TableDef
Dim db As DAO.Database
Dim fldTemp As DAO.Field

Set db = CodeDb
Open "C:\Temp\types.txt" For Output As #1
Print #1, "'**************types for tables***********"
Print #1, "'used by forms as temp tables"
Print #1, "'"

For Each tblTemp In db.TableDefs
    Print #1, "Type T" & tblTemp.Name
    
    For Each fldTemp In tblTemp.Fields
        Select Case fldTemp.Type
          Case dbText, dbMemo, dbComplexText
               Print #1, fldTemp.Name & " as String"
          Case dbDate
               Print #1, fldTemp.Name & " as Date"
          Case dbNumeric, dbDouble, dbInteger, dbLong
               Print #1, fldTemp.Name & " as Integer"
        End Select
    Next fldTemp
    
    Print #1, "End Type"
    Print #1, " "
Next tblTemp

Set db = Nothing
Set tblTemp = Nothing
Set fldTemp = Nothing
Print #1, "'************end types for tables*********"
Close #1
MsgBox "Created!"
End Sub

Function IsNothingIn(vVal As Variant) As Boolean
'Created 21/08/2024 By Roger Williams
'
'checks if vartemp is empty
'
On Error Resume Next

IsNothingIn = False

If IsEmpty(vVal) Then
   IsNothingIn = True
   Exit Function
End If

If IsNull(vVal) Then
   IsNothingIn = True
   Exit Function
End If

If Len(vVal) = 0 Then
   IsNothingIn = True
   Exit Function
End If

End Function

Sub ListTablesandFieldsAsJavaScript()
'lists tables to: c:\temp\tablesforjavascript.txt as JavaScript objects
Dim tblTemp As DAO.TableDef
Dim db As DAO.Database
Dim fldTemp As DAO.Field
Dim intNum As Integer
Dim strPrimary As String

Set db = CodeDb
Open "C:\Temp\tablesforjavascript.txt" For Output As #1
Print #1, "/*"
Print #1, "  Note: IndexedDB does not support field sizes they are included for HTML"
Print #1, "        data entry validation"
Print #1, "*/"
Print #1, " "
Print #1, "//**************tables schema************"
Print #1, Chr(34) & "use strict" & Chr(34)
Print #1, " "
Print #1, "export const aryTables = ["
    
For Each tblTemp In db.TableDefs
   If InStr(1, tblTemp.Name, "msys") = 0 And InStr(1, tblTemp.Name, "usys") = 0 And InStr(1, tblTemp.Name, "ribbon") = 0 Then

    Print #1, "  {"
    Print #1, "tblName: " & Chr(34) & tblTemp.Name & Chr(34) & ","
    Print #1, "aryFields: ["
    
    'get primary key
    For intNum = 0 To tblTemp.Indexes.Count - 1
         If tblTemp.Indexes(intNum).Primary Then
            strPrimary = tblTemp.Indexes(intNum).Fields(0).Name
            Exit For
         End If
    Next intNum
    
    For Each fldTemp In tblTemp.Fields
        Print #1, " {"
        Print #1, "fieldName: " & Chr(34) & fldTemp.Name & Chr(34) & ","
         
        If fldTemp.Name = tblTemp.Fields(0).Name Then
          Print #1, "primaryKey: true,"
        Else
          Print #1, "primaryKey: false,"
        End If
        
        If fldTemp.Type = 4 Then
            Print #1, "autoNumber: true,"
        Else
            Print #1, "autoNumber: false,"
        End If
    
        Select Case fldTemp.Type
          Case 8
             Print #1, "fieldSize: 12,"
             'add empty field value so web applioation can use it for data storage
             Print #1, "fieldValue: " & Chr(34) & Chr(34) & ","
          Case 10
             Print #1, "fieldSize: " & fldTemp.Size & ","
             'add empty field value so web applioation can use it for data storage
             Print #1, "fieldValue: " & Chr(34) & Chr(34) & ","
         Case 12
             Print #1, "fieldSize: 4096,"  'memo field
             'add empty field value so web applioation can use it for data storage
             Print #1, "fieldValue: " & Chr(34) & Chr(34) & ","

          Case Else
             Print #1, "fieldSize: 0,"
             'add empty field value so web applioation can use it for data storage
             Print #1, "fieldValue: 0,"

        End Select
        
        If fldTemp.Required Then
           Print #1, "requiredField: true"
        Else
           Print #1, "requiredField: false"
        End If
        
        Print #1, "    },"
    Next fldTemp
    

    Print #1, "    ],"
    Print #1, "    },"
    
    Print #1, vbCrLf
 End If
Next tblTemp
    
Print #1, "    ];"
Print #1, "//************end tables*********"
Close #1
MsgBox "Finished!"
End Sub

Sub deleteforms()
'Created 27/06/2016 By Roger Williams
'
'Deletes all forms in the database
'

Dim docForm As Document
Dim colTemp As Container
Dim db As DAO.Database
Dim frmTemp As Form

Set db = CodeDb
Set colTemp = db.Containers("Forms")

For Each docForm In colTemp.Documents
    DoCmd.DeleteObject acForm, docForm.Name
Next docForm

db.Close
Set colTemp = Nothing
MsgBox "Form Delete Complete COMPACT This Database", vbInformation

End Sub
Sub exportformstotext(strCycle As String)
'Created 27/06/2016 By Roger Williams
'
'exports all forms to text files, creates directory to contain the text files
'if it does not exist
'
'VARS
'
'strCycle: cycle e.g. Sell
'

Dim docForm As Document
Dim colTemp As Container
Dim db As DAO.Database

If strCycle = "" Then
   MsgBox "No Cycle Name Supplied e.g. sell", vbCritical, "Error"
   Exit Sub
End If

If Dir$("c:\" & strCycle, vbDirectory) = "" Then
   MkDir "c:\" & strCycle
End If

If Dir$("c:\" & strCycle & "\Forms", vbDirectory) = "" Then
   MkDir "c:\" & strCycle & "\Forms"
End If

Set db = CodeDb
Set colTemp = db.Containers("Forms")

For Each docForm In colTemp.Documents
    SaveAsText acForm, docForm.Name, "c:\" & strCycle & "\Forms\" & docForm.Name & ".txt"
Next docForm

db.Close
Set colTemp = Nothing
MsgBox "Export To Text Complete", vbInformation
End Sub
Sub importformsfromtext(strCycle As String)
'Created 27/06/2016 By Roger Williams
'
'imports all forms from text files created in a directory named by strCycle
'
'VARS
'
'strCycle: cycle e.g. Sell where the text forms as stored
'
'NOTE: user must first run deleteforms BEFORE running this as the database
'needs to be compacted after deletion of existing forms.
'

Dim strTemp As String
Dim strForm As String

If strCycle = "" Then
   MsgBox "No Cycle Name Supplied e.g. sell", vbCritical, "Error"
   Exit Sub
End If

If Dir$("c:\" & strCycle, vbDirectory) = "" Then
   MsgBox strCycle & " Does Not Exist", vbExclamation, "No Forms To Import"
End If

strTemp = Dir$("c:\" & strCycle & "\Forms\*.txt")

While strTemp <> ""
    strForm = Mid$(strTemp, 1, InStr(1, strTemp, ".") - 1)
    LoadFromText acForm, strForm, "c:\" & strCycle & "\Forms\" & strTemp
    strTemp = Dir$
Wend

MsgBox "Import From Text Complete", vbInformation
End Sub
'*******************             end of forms export/import/delete           *******************
'***********************************************************************************************

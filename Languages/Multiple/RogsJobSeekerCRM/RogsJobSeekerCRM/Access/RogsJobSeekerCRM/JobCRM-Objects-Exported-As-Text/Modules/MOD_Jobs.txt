Sub PopulateFormLabels(strTable As String, ByRef frmForm As Form)
'Created 21/08/2024 By Roger Williams
'
'uses the descripton property of the fields in the passed table schema to set the forms
'label captions
'
'does this by stripping the controls name string starting at the first _
'this is used to determine the label is linked to a field e.g. LBL_JOB_Title
'if the field has no _ it is ignored
'
'for textboxes, checkboxes, listboxes and combos changes controls TAG to field type
'if not numeric then sets tag with delimeter : to include field length
'this is used when saving
'
'NOTE: dates have tag added of :10
'
Dim tblTemp As DAO.TableDef
Dim fldTemp As DAO.Field
Dim ctlTemp As control
Dim db As DAO.Database
Dim strTemp As String
Dim intNum As Integer
On Error GoTo ERR_Error

Set db = CodeDb
Set tblTemp = db.TableDefs(strTable)

'For Each fldTemp In tblTemp.Fields
    For Each ctlTemp In frmForm.Controls
        intNum = InStr(1, ctlTemp.Name, "_")
        
        If intNum <> 0 Then
            'extract field name
           strTemp = Mid$(ctlTemp.Name, intNum + 1, Len(ctlTemp.Name))
           
           If tblTemp.Fields(strTemp).Properties("Description") <> "" Then
              If ctlTemp.ControlType = acLabel Then
                 ctlTemp.Caption = tblTemp.Fields(strTemp).Properties("Description") & ":"
                 'make sure label is wide enough!
                 ctlTemp.Width = (Len(tblTemp.Fields(strTemp).Properties("Description")) * 140) 'convert cm to twips
              Else
                 'set tag to field type
                 ctlTemp.Tag = tblTemp.Fields(strTemp).Type
                 
                 'if applicable add to tag field length
                 Select Case tblTemp.Fields(strTemp).Type
                     Case dbText
                          ctlTemp.Tag = ctlTemp.Tag & ":" & tblTemp.Fields(strTemp).Size
                     Case dbDate
                          ctlTemp.Tag = ctlTemp.Tag & ":10"
                 End Select
              End If
         End If
        End If
    Next ctlTemp
'Next fldTemp
Set tblTemp = Nothing
Set ctlTemp = Nothing
Set db = Nothing
Exit Sub

ERR_Error:
'skip if description ssing or missing caption proerty
If Err.Number = 91 Or Err.Number = 3265 Or Err.Number = 3270 Then Resume Next
Stop

End Sub
Function ValidateInputLength(ctlTemp As control, intKey As Integer) As Integer
'Created 23/08/2024 By Roger Williams
'
'makes sure entered text is NOT longer than field size
'
'passed controls tag is read for value after :
'
'e.g.:
'
'ctltemp.tag = 1-:25  - max cahrs allowed 24
'
'if text value of control is > max length returns ZERO this should cancel
'any pressed key from being added to the controls text
'else returns key pressed
'
'
Dim strTemp As String
'Stop
'extract max size
strTemp = Mid$(ctlTemp.Tag, InStr(1, ctlTemp.Tag, ":") + 1, Len(ctlTemp.Tag))

If Len(ctlTemp.Value) > CInt(strTemp) Then
   ValidateInputLength = 0
Else
   ValidateInputLength = intKey
End If
End Function

Sub InitForm(ByRef frmTemp As Form, strFieldPrefix As String)
'Modified 23/08/2024 By Roger Williams
'
'Originally run in each form now moved here for global use
'hence now passed form and field name which is the field name only
'as far as the _ e.g. TYP_, STA_ etc.
'
'
'Created 21/08/2024 By Roger Williams
'
'resets form data entry controls
'
'rather than hardcoding the controls names here looks through the control name
'to see if they are data fields i.e. the name has a filed name at the end of it
'e.g. TXT_JOB_Title
'
'this means developer can add new fields to the table and form and not need to
'touch this code!
'

Dim ctlTemp As control
Dim strTemp As String
On Error GoTo ERR_Error

'go through each control and add its value to the query IF table holds field data
For Each ctlTemp In frmTemp.Controls
    If InStr(1, ctlTemp.Name, strFieldPrefix) <> 0 And Not IsNothingIn(ctlTemp.Tag) Then
       'extract field type from control tag
       If InStr(1, ctlTemp.Tag, ":") <> 0 Then
          strTemp = Mid$(ctlTemp.Tag, 1, InStr(1, ctlTemp.Tag, ":") - 1)
       Else
          strTemp = ctlTemp.Tag
       End If
       
       If strTemp <> vbNullString Then
       'get field type
           Select Case CInt(strTemp)
              Case dbText, dbMemo, dbComplexText
                   ctlTemp.Value = vbNullString
              Case dbDate
                   ctlTemp.Value = vbNullString
              Case dbNumeric, dbDouble, dbInteger, dbLong
                   ctlTemp.Value = 0
           End Select
       End If
    End If
Next ctlTemp

Exit Sub
ERR_Error:
Stop
Resume Next
End Sub

Function SaveData(frmTemp As Form, strFieldPrefix As String, strTable As String, lngID As Long) As Integer
'Created 23/08/2024 By Roger Williams
'
'MOVED from forms CMDSave_Click
'
'Added parameters:
'
'frmTemp        - form to read data from
'strFieldPrefix - field prefix e.g. JOB_
'strTable       - table e.g. Seekers_Jobs
'lngID          - from typ<table>old.<fieldprefix>ID
'
'
'saves the contents of the type to the table
'
'rather than hardcoding the controls names here looks through the control name
'to see if they are data fields i.e. the name has a field name at the end of it
'e.g. TXT_JOB_Title
'
'this means developer can add new fields to the table and form and not need to
'touch this code!
'
Dim strInsertValues     As String
Dim strInsertFields     As String
Dim strUpdateValues     As String
Dim strSQL              As String
Dim strSQLLines         As String
Dim strFieldName        As String
Dim strTemp             As String
Dim intNum              As Integer
Dim ctlTemp             As control
Dim db                  As DAO.Database

Set db = CodeDb
On Error GoTo ERR_Error

'determine if new record
If lngID = 0 Then
   strInsertFields = "INSERT INTO " & strTable & " ("
   strInsertValues = "VALUES ("
   'go through each control and add its value to the query IF table holds field data
   For Each ctlTemp In frmTemp.Controls
       If ctlTemp.ControlType = acTextBox Or ctlTemp.ControlType = acComboBox Or ctlTemp.ControlType = acCheckBox Then
            If InStr(1, ctlTemp.Name, strFieldPrefix) <> 0 Then
               'get field name
               strFieldName = Mid$(ctlTemp.Name, InStr(1, ctlTemp.Name, strFieldPrefix), Len(ctlTemp.Name))
               strInsertFields = strInsertFields & strFieldName & ","
               'extract field type from control tag
               If InStr(1, ctlTemp.Tag, ":") = 0 Then
                  strTemp = ctlTemp.Tag
               Else
                  strTemp = Mid$(ctlTemp.Tag, 1, InStr(1, ctlTemp.Tag, ":") - 1)
               End If
               
               Select Case CInt(strTemp)
                  Case dbText, dbMemo, dbComplexText
                       strInsertValues = strInsertValues & Chr(34) & Nz(ctlTemp.Value, vbNullString) & Chr(34) & ","
                  Case dbDate
                       strInsertValues = strInsertValues & IIf(Len(ctlTemp.Value) = 0, "Null", "#" & Format(ctlTemp.Value, "mm/dd/yyyy") & "#") & ","
                  Case dbNumeric, dbDouble, dbInteger, dbBoolean, dbLong
                       strInsertValues = strInsertValues & Nz(ctlTemp.Value, 0) & ","
               End Select
            End If
       End If
NextControl:
   Next ctlTemp
   
   'strip end ,
   strInsertFields = Mid$(strInsertFields, 1, Len(strInsertFields) - 1)
   strInsertValues = Mid$(strInsertValues, 1, Len(strInsertValues) - 1)
   strSQL = strInsertFields & ") " & strInsertValues & ");"
Else
   'update data
   strUpdateValues = "UPDATE " & strTable & " SET "
   'go through each control and add its value to the query IF table holds field data
   For Each ctlTemp In frmTemp.Controls
       If ctlTemp.ControlType = acTextBox Or ctlTemp.ControlType = acComboBox Or ctlTemp.ControlType = acCheckBox Then
            If InStr(1, ctlTemp.Name, strFieldPrefix) <> 0 Then
               'get field name
               strFieldName = Mid$(ctlTemp.Name, InStr(1, ctlTemp.Name, strFieldPrefix), Len(ctlTemp.Name))
               'extract field type from control tag
               
               If InStr(1, ctlTemp.Tag, ":") = 0 Then
                  strTemp = ctlTemp.Tag
               Else
                  strTemp = Mid$(ctlTemp.Tag, 1, InStr(1, ctlTemp.Tag, ":") - 1)
               End If
               
               Select Case CInt(strTemp)
                  Case dbText, dbMemo, dbComplexText
                       strUpdateValues = strUpdateValues & strFieldName & " = " & Chr(34) & Nz(ctlTemp.Value, vbNullString) & Chr(34) & ","
                    Case dbDate
                       strUpdateValues = strUpdateValues & strFieldName & " = " & IIf(Nz(ctlTemp.Value, 0) = 0, "Null", "#" & Format(ctlTemp.Value, "mm/dd/yyyy") & "#") & ","
                  Case dbNumeric, dbDouble, dbInteger, dbLong, dbBoolean
                       strUpdateValues = strUpdateValues & strFieldName & " = " & Nz(ctlTemp.Value, 0) & ","
                End Select
            End If
       End If
   Next ctlTemp
       
   'strip end ,
   strUpdateValues = Mid$(strUpdateValues, 1, Len(strUpdateValues) - 1)
   strSQL = strUpdateValues & " WHERE " & strFieldPrefix & "ID=" & lngID & ";"
End If

'save job data
db.Execute strSQL, dbFailOnError
frmTemp.LBLStatus.Caption = "Saved"
SaveData = 0
MsgBox "Data Saved!", vbInformation, "Changes Saved"
Exit Function

ERR_Error:
Stop
Resume Next
SaveData = Err.Number
MsgBox "Unexpected Error", vbExclamation, "Error"
Resume NextControl

End Function
